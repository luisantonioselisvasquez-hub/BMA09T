// Fibonacci con tabulación 
#include <stdio.h>

int main() {
    int n;
    printf("Ingrese n: ");
    scanf("%i", &n);

    // 1. Declaramos la tabla (arreglo)
    int tabla[100];

    // 2. Casos base
    tabla[0] = 0;
    tabla[1] = 1;

    // 3. Llenamos la tabla de abajo hacia arriba
    int i;
    for (i = 2; i <= n; i++) {
        tabla[i] = tabla[i - 1] + tabla[i - 2];
    }

    // 4. La respuesta está en tabla[n]
    printf("Fibonacci(%i) = %i\n", n, tabla[n]);

    return 0;
}
