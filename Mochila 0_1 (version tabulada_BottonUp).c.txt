/*Ejemplo clásico: Mochila 0/1 (versión tabulada)
Dado un conjunto de objetos con peso y valor, y una mochila de capacidad W, maximizar el valor sin exceder el peso.*/

#include <stdio.h>

#define MAX_OBJ 100
#define MAX_CAP 1000

int max(int a, int b) { return (a > b) ? a : b; }

int main() {
    int n = 4;                              // cantidad de objetos
    int peso[]  = {0, 1, 3, 4, 5};          // (índice 0 no se usa)
    int valor[] = {0, 1, 4, 5, 7};
    int W = 7;                              // capacidad

    // tabla[i][w] = mejor valor usando los primeros i objetos con capacidad w
    int tabla[MAX_OBJ][MAX_CAP];

    // Caso base: con 0 objetos o capacidad 0, el valor es 0
    int i,w;
    for ( w = 0; w <= W; w++) tabla[0][w] = 0;
    for ( i = 0; i <= n; i++) tabla[i][0] = 0;

    // Llenado bottom-up
    for ( i = 1; i <= n; i++) {
        for ( w = 1; w <= W; w++) {
            if (peso[i] <= w) {
                // Opción 1: no llevo el objeto i  -> tabla[i-1][w]
                // Opción 2: sí lo llevo            -> valor[i] + tabla[i-1][w - peso[i]]
                tabla[i][w] = max(tabla[i-1][w],
                                  valor[i] + tabla[i-1][w - peso[i]]);
            } else {
                // No cabe, solo puedo no llevarlo
                tabla[i][w] = tabla[i-1][w];
            }
        }
    }

    printf("Valor maximo: %d\n", tabla[n][W]);
    return 0;
}
/*Aquí la tabla es 2D: tabla[i][w] guarda la mejor solución usando los primeros i objetos con 
capacidad w. Se va construyendo desde (0,0) hasta (n, W).*/
